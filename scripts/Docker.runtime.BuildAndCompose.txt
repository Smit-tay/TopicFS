/////////////////////////

///////////  BUILD topicfs_runtime image /////////////

cd /home/jack/dev/smithjack.net/topicfs

sudo podman build \
  --format docker \
  -f Dockerfile.runtime \
  -t localhost/topicfs_runtime:latest \
  .
  

///////////  COMPOSE (and run) topicfs_runtime image /////////////
sudo podman run -d \
  --name topicfs_runtime \
  --hostname topicfs-runtime \
  --network host \
  --privileged \
  --device /dev/fuse:/dev/fuse \
  --security-opt label=disable \
  --cap-add SYS_ADMIN \
  -e TZ=Europe/Berlin \
  -e ROS_DOMAIN_ID=11 \
  -e TOPICFS_MOUNT_POINT=/home/jack/fuse_mount \
  -v /home/jack/dev/smithjack.net/topicfs/install:/home/jack/dev/smithjack.net/topicfs/install:z \
  -v /home/jack/fuse_mount:/home/jack/fuse_mount:z,shared \
  -v /opt/topicfs_typesupport:/opt/topicfs_typesupport:z \
  localhost/topicfs_runtime:latest
